package proyecto_entornos;
public class ProductoFisico extends Producto //HERENCIA
{
    /*ATRIBUTOS*/
    private double costeEnvio; 
    
    /*CONTRUSTOR*/
    public ProductoFisico(String nombre, double precio, double costeEnvio)
    {
        super(nombre, precio);
        if (costeEnvio < 0) {
            throw new IllegalArgumentException("El coste de envío no puede ser negativo.");
        }
        this.costeEnvio = costeEnvio;
    }
    
    /*METODOS GET Y SET*/
    //COSTE ENVIO
    public double getCosteEnvio()
    {
        return this.costeEnvio;
    }
    public void setCosteEnvio(double costeEnvio)
    {
        this.costeEnvio = costeEnvio;
    }
    
    /* METODOS */
    @Override
    public double calcularPrecioFinal()
    {
        return getPrecio() + this.costeEnvio;
    }


    //toString
    @Override
    public String toString()
    {
        String salida = super.toString();
        salida += "Coste de envio: " + this.costeEnvio;
        return salida;
    }
}
